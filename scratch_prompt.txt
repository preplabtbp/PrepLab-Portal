function bacaRosterSheet(ss, sheetName, kolom) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];

  var hasil = [];
  for (var r = 2; r < data.length; r++) {
    var nama = data[r][kolom.nama];

function getISOWeekNumber(d) {
  var date = new Date(d.getTime());
  date.setHours(0, 0, 0, 0);
  date.setDate(date.getDate() + 3 - (date.getDay() + 6) % 7);
  var week1 = new Date(date.getFullYear(), 0, 4);
  var weekNum = 1 + Math.round(((date.getTime() - week1.getTime()) / 86400000 - 3 + (week1.getDay() + 6) % 7) / 7);
  return { year: date.getFullYear(), week: ("0" + weekNum).slice(-2) };
}

function sisipkanGambarCerdasGlobal(body, tag, base64Data, maxW, maxH) {
  var searchResult = body.findText(tag);
  if (!searchResult) return;
  var textElement = searchResult.getElement().asText();
  if (!base64Data || base64Data.length < 100 || base64Data === "-") { textElement.setText(textElement.getText().replace(tag, ""));
  return; }
  try {
    var b64str = base64Data; var type = "image/png";
    if (b64str.indexOf(",") !== -1) { type = b64str.split(';')[0].replace('data:', ''); b64str = b64str.split(',')[1];
    }
    var blob = Utilities.newBlob(Utilities.base64Decode(b64str), type);
    textElement.setText(textElement.getText().replace(tag, ""));
    var img = textElement.getParent().asParagraph().appendInlineImage(blob);
    var scaleFactor = Math.min(maxW / img.getWidth(), maxH / img.getHeight(), 1);
    img.setWidth(img.getWidth() * scaleFactor); img.setHeight(img.getHeight() * scaleFactor);
  } catch (e) { textElement.setText(textElement.getText().replace(tag, "")); }
}

      dataF[j][19] = dataF[j][19] || "-"; dataF[j][20] = dataF[j][20] || "-";
      dataF[j][21] = dataF[j][21] || "-"; 
    }

    if (devOpt.pdf === false) {
        addLog("PDF_BYPASS", "Pembuatan PDF dilewati karena Dev Mode");

      }
      return JSON.stringify({ status: "error", message: err.message });
  }

function buatPDFOtomatis(data, folder, ttd1, ttd2, ttd3, fotoProses, templateId, suffixNama) {
  var header = data[0];
  var rawJam = header[0]; var rawTgl = header[1];
  
  if (rawJam instanceof Date) rawJam = Utilities.formatDate(rawJam, "Asia/Jayapura", "HH:mm");
  if (rawTgl instanceof Date) rawTgl = Utilities.formatDate(rawTgl, "Asia/Jayapura", "dd/MM/yyyy");
  
  var jamString = rawJam ? rawJam.toString() : "-";
  var tglString = rawTgl ? rawTgl.toString() : "-";
  var bagianString = header[2] ? header[2].toString() : "-";
  var bagian = bagianString.replace(/\//g, "-");
  
  var isoWeek = getISOWeekNumber(new Date());
  var prefixWeek = "W" + isoWeek.week + "_" + isoWeek.year;
  var namaFileBaru = prefixWeek + "_APD_" + bagian + suffixNama;

  var docFile = DriveApp.getFileById(templateId).makeCopy(namaFileBaru, folder);
  var doc = DocumentApp.openById(docFile.getId()); var body = doc.getBody();

  body.replaceText("<<Jam>>", jamString); body.replaceText("<<Tanggal>>", tglString);
  body.replaceText("<<Bagian>>", bagianString); body.replaceText("<<Waktu Shift>>", header[3] || "-");
  body.replaceText("<<Nama Insp 1>>", header[16] || "-"); body.replaceText("<<Jabatan Insp 1>>", header[17] || "-");
  body.replaceText("<<Nama Insp 2>>", header[18] || "-");
  body.replaceText("<<Jabatan Insp 2>>", header[19] || "-");
  body.replaceText("<<Nama Insp 3>>", header[20] || "-"); body.replaceText("<<Jabatan Insp 3>>", header[21] || "-");
  var tables = body.getTables();
  if(tables.length > 0) {
    var mainTable = tables[0]; var barisTag = mainTable.getRow(1).copy(); mainTable.removeRow(1);
    for(var i = 0; i < data.length; i++) {
      var rowData = data[i];
      var newRow = barisTag.copy();
      newRow.replaceText("<<No>>", rowData[5]); newRow.replaceText("<<Nama>>", rowData[6]); newRow.replaceText("<<Jabatan>>", rowData[7]);
      newRow.replaceText("<<Kehadiran>>", rowData[8]); newRow.replaceText("<<Seragam>>", rowData[9]); newRow.replaceText("<<Helm>>", rowData[10]);
      newRow.replaceText("<<Sepatu>>", rowData[11]); newRow.replaceText("<<Masker>>", rowData[12]);
      newRow.replaceText("<<Ear Plug>>", rowData[13]);
      newRow.replaceText("<<Kacamata>>", rowData[14]); newRow.replaceText("<<Keterangan>>", rowData[15]); mainTable.appendTableRow(newRow); 
    }
  }

  sisipkanGambarCerdasGlobal(body, "<<TTD 1>>", ttd1, 150, 100);
  sisipkanGambarCerdasGlobal(body, "<<TTD 2>>", ttd2, 150, 100);
  sisipkanGambarCerdasGlobal(body, "<<TTD 3>>", ttd3, 150, 100); 
  
  sisipkanGambarCerdasGlobal(body, "<<Foto Proses Inspeksi>>", fotoProses, 350, 350);
  sisipkanGambarCerdasGlobal(body, "<<Foto Proses>>", fotoProses, 350, 350);

  doc.saveAndClose();

  Utilities.sleep(4000);

  var pdfBlob = doc.getAs('application/pdf'); var pdfFile = folder.createFile(pdfBlob).setName(namaFileBaru + ".pdf");
  pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); docFile.setTrashed(true);
  return "https://drive.google.com/uc?export=view&id=" + pdfFile.getId();
}

function cariBarisKaryawanRooster(ss, nama) {
  var konfigSheet = [
    { sheetName: "Rooster_Staff", kolomNama: 1, kolomShift: 8 }, 
    { sheetName: "Rooster_Crew",  kolomNama: 1, kolomShift: 7 }  
  ];
  var namaBersih = nama.toString().trim();

  for (var k = 0; k < konfigSheet.length; k++) {
    var sheet = ss.getSheetByName(konfigSheet[k].sheetName);
    if (!sheet) continue;
    var data = sheet.getDataRange().getValues();
    for (var r = 2; r < data.length; r++) {
      var namaCell = data[r][konfigSheet[k].kolomNama];
      if (namaCell && namaCell.toString().trim() === namaBersih) {
        return { sheet: sheet, row: r + 1, kolomShift: konfigSheet[k].kolomShift };
      }
    }
  }
  return null;
}

      buktiSelesai: bersihkanData(row[11], false),  
      picClosing: bersihkanData(row[12], false)     
    };

}

function closeJobTicket(ticketId, namaPic, fotoBase64, deskripsiPerbaikan, devOpt) {
  var startTime = new Date().getTime();
  var isDev = devOpt && devOpt.isDev;
  var logs = { steps: [], timings: {}, status: "success" };

  function addLog(action, detail) { if(isDev && devOpt.verboseLog) logs.steps.push({ time: (new Date().getTime() - startTime) + "ms", action: action, info: detail }); }

  try {
    addLog("INIT", "Memulai proses closing tiket " + ticketId);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Rekap_Temuan");
    if (!sheet) throw new Error("Sheet Rekap_Temuan tidak ditemukan.");
    
    var data = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < data.length; i++) {
      if (data[i][1] === ticketId) { rowIndex = i + 1; break; }
    }
    if (rowIndex === -1) throw new Error("Ticket ID tidak ditemukan.");
    
    var urlFoto = "-";
    if (fotoBase64 && fotoBase64.length > 100) {
       addLog("UPLOAD", "Mengunggah foto bukti perbaikan");
       var folder = DriveApp.getFolderById(FOLDER_ID);
       var splitBase = fotoBase64.split(',');
       var type = splitBase[0].split(';')[0].replace('data:', '');
       var blob = Utilities.newBlob(Utilities.base64Decode(splitBase[1]), type, "CLOSING_" + ticketId + ".png");
       var file = folder.createFile(blob);
       file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
       urlFoto = "https://drive.google.com/uc?export=view&id=" + file.getId();
    }
    
    var now = new Date();
    var tglSelesai = Utilities.formatDate(now, "Asia/Jayapura", "dd/MM/yyyy HH:mm");
    
    if (isDev && devOpt.db === false) {
        addLog("DB_BYPASS", "Proses update status di Spreadsheet dilewati (Dev Mode)");
    } else {
        addLog("DB_UPDATE", "Memperbarui status tiket ke CLOSED di Sheet");
        sheet.getRange(rowIndex, 9).setValue("CLOSED");            
        sheet.getRange(rowIndex, 10).setValue(tglSelesai);         
        
        sheet.getRange(rowIndex, 12).setValue(urlFoto);            
        sheet.getRange(rowIndex, 13).setValue(namaPic);            
        sheet.getRange(rowIndex, 14).setValue(deskripsiPerbaikan); 
    }

    logs.timings.totalExecutionTime = (new Date().getTime() - startTime) + "ms";

    if (isDev && devOpt.verboseLog) {
        return { isVerbose: true, status: "success", logDetails: logs };
    }
    return "Tiket " + ticketId + " berhasil ditutup!";

  } catch (e) {
    if (isDev && devOpt.verboseLog) {
        logs.status = "error";
        logs.errorMessage = e.message;
        return { isVerbose: true, status: "error", logDetails: logs };
    }
    throw new Error(e.message);
  }
}

function tentukanSection(area) {
  var a = area.toString().toUpperCase();

  for (var i = 0; i < MASTER_29_INSPEKSI.length; i++) {
    var master = MASTER_29_INSPEKSI[i];
    if (a.indexOf(master.nama.toUpperCase()) !== -1 || 
        master.nama.toUpperCase().indexOf(a) !== -1) {
      
      var kw = Array.isArray(master.kw) ? master.kw[0] : master.kw;
      if (MAP_KW_TO_SECTION[kw]) return MAP_KW_TO_SECTION[kw];
    }
  }

  var a2 = area.toString().toUpperCase();
  if (a2.indexOf("SAMPEL") !== -1)      return "PREPARATION"; 
  if (a2.indexOf("QA") !== -1)          return "QA";
  if (a2.indexOf("GUDANG") !== -1)      return "IC";
  if (a2.indexOf("LAB") !== -1 || 
      a2.indexOf("XRF") !== -1 || 
      a2.indexOf("FUSION") !== -1)      return "LABORATORY";
  if (a2.indexOf("PREP") !== -1 || 
      a2.indexOf("BASAH") !== -1 || 
      a2.indexOf("KERING") !== -1 || 
      a2.indexOf("PREPARASI") !== -1)   return "PREPARATION";
  if (a2.indexOf("MAINTENANCE") !== -1 || 
      a2.indexOf("WORKSHOP") !== -1)    return "MAINTENANCE";

  return null;
}

function generateTicketID(offset) {
  offset = offset || 0; 
  var now = new Date();
  
  var d = new Date(Date.UTC(now.getFullYear(), now.getMonth(), now.getDate()));
  var dayNum = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  var yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  var weekNo = Math.ceil((((d - yearStart) / 86400000) + 1) / 7);
  
  var yearYY = now.getFullYear().toString().slice(-2);
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetRekap = ss.getSheetByName("Rekap_Temuan");
  
  var newCounter = 1; 
  
  if (sheetRekap) {
    var lastRow = sheetRekap.getLastRow();
    if (lastRow > 0) {
      newCounter = lastRow + offset; 
    }
  }
  
  var counterStr = newCounter.toString();
  while (counterStr.length < 3) counterStr = "0" + counterStr;
  
  return "TKT-W" + weekNo + "Y" + yearYY + "-" + counterStr;
}

      var tglRaw, rowDate, customArea, uniqueKey;
      var colD = data[r][3] ? data[r][3].toString().trim() : ""; 
      
      if (conf.name === "Log_Umum") {
        tglRaw = data[r][0]; 
        customArea = data[r][cAreaUmum] ? data[r][cAreaUmum].toString().trim() : "Area Umum";
        uniqueKey = tglRaw ? tglRaw.toString() + "_" + customArea : null; 
      }
      else if (conf.name === "Log_APD") {
        tglRaw = data[r][1]; 
        var colC = data[r][2] ? data[r][2].toString().trim() : "";
        customArea = "Inspeksi APD (" + colC + ")";
        uniqueKey = tglRaw ? tglRaw.toString() + "_" + customArea : null; 
      }
      else if (conf.name === "Log_P3K") {
        tglRaw = data[r][1]; 
        var colE = data[r][4] ? data[r][4].toString().trim() : "";
        customArea = "Inspeksi Kotak P3K (" + colE + ")";
        uniqueKey = tglRaw ? tglRaw.toString() + "_" + customArea : null;
      }
      else if (conf.name === "Log_Perkakas") {
        tglRaw = data[r][0]; 
        customArea = "Inspeksi Perkakas Tangan Portable";
        uniqueKey = colD || (tglRaw ? tglRaw.toString() : null); 
      }
      else if (conf.name === "Log_Tabung") {
        tglRaw = data[r][0]; 
        customArea = "Inspeksi Harian Pra Pakai Tabung Gas Bertekanan";
        uniqueKey = colD || (tglRaw ? tglRaw.toString() : null); 
      }
      else if (conf.name === "Log_Tangga") {
        tglRaw = data[r][0]; 
        customArea = "Inspeksi Tangga Portabel";
        uniqueKey = colD || (tglRaw ? tglRaw.toString() : null); 
      }
      else if (conf.name === "Log_Sarana") {
        tglRaw = data[r][0];
        customArea = "Inspeksi Sarana Prasarana Unit";
        uniqueKey = colD || (tglRaw ? tglRaw.toString() : null); 
      }
      
      if (!uniqueKey || trackedIds[uniqueKey]) continue; 
      
      if (tglRaw instanceof Date) {
        rowDate = tglRaw;
      } else if (typeof tglRaw === "string" && tglRaw.indexOf("/") !== -1) {
         var parts = tglRaw.split("/");
         if(parts.length === 3) {
             var h = parseInt(parts[0], 10);
             var b = parseInt(parts[1], 10) - 1;
             var t = parseInt(parts[2], 10);
             if (t < 100) t += 2000;
             rowDate = new Date(t, b, h);
         }
      } else {
         rowDate = new Date(tglRaw);
      }
      
      if (rowDate && rowDate >= mulaiMingguLalu) {
        
        var fotoDitemukan = false;

        for (var u = 0; u < cUrlArray.length; u++) {
            var url = data[r][cUrlArray[u]];
            
            if (url && url !== "-" && url.toString().indexOf("http") !== -1) {
              fotoDitemukan = true;
              
              var weekLabel = "";
              if (conf.name === "Log_Perkakas" || conf.name === "Log_Tabung" || conf.name === "Log_Tangga" || conf.name === "Log_Sarana") {
                 var matchW = colD.match(/[Ww]\d+_\d+/); 
                 if (matchW) {
                     weekLabel = matchW[0].toUpperCase().replace("_", "-"); 
                 } else {
                     weekLabel = "W" + getISOWeekNumber(rowDate).week + "-" + getISOWeekNumber(rowDate).year;
                 }
              } else {
                 weekLabel = "W" + getISOWeekNumber(rowDate).week + "-" + getISOWeekNumber(rowDate).year;
              }
              
              var suffixArea = "";
              if(conf.name === "Log_Sarana") {
                  var namaUnit = data[r][4] ? data[r][4].toString().trim() : ("Unit " + (u+1));
                  suffixArea = " (" + namaUnit + ")";
              }

              hasil.push({
                tanggal: Utilities.formatDate(rowDate, "Asia/Jayapura", "dd/MM/yyyy HH:mm"),
                week: weekLabel,
                url: url.toString().trim(),
                inspektor: data[r][cInsp] ? data[r][cInsp].toString().split(" | ")[0].trim() : "-",
                area: customArea + suffixArea,
                sumber: conf.name.replace("Log_", "")
              });
            }
        }
        
        if (fotoDitemukan) trackedIds[uniqueKey] = true; 
      }
    }
  }
  
  hasil.sort(function(a, b) {
    var da = a.tanggal.split(" ")[0].split("/"); 
    var db = b.tanggal.split(" ")[0].split("/");
    var ta = a.tanggal.split(" ")[1].split(":");
    var tb = b.tanggal.split(" ")[1].split(":");
    var timeA = ta.length === 2 ? ta : ["00", "00"];
    var timeB = tb.length === 2 ? tb : ["00", "00"];
    var dateA = new Date(da[2], da[1]-1, da[0], timeA[0], timeA[1]);
    var dateB = new Date(db[2], db[1]-1, db[0], timeB[0], timeB[1]);
    return dateB - dateA;
  });
  
  return hasil;
}

function doPost(e) {
  try {
    var rawData = e.postData.contents;
    var payload = JSON.parse(rawData);
    var action = payload.action;
    
    var result = null;

    switch (action) {
      case "getMasterPertanyaan":
        result = getMasterPertanyaan();
        break;

      case "submitInspeksiUniversal":
        result = submitInspeksiUniversal(payload.finalData, payload.ttd1, payload.ttd2, payload.ttd3, payload.fotoTemuanArray, payload.fotoProses);
        break;

      case "submitInspeksi":
        result = submitInspeksi(payload.dataF, payload.ttd1, payload.ttd2, payload.ttd3, payload.fotoProses, payload.devOptions);
        break;
      case "getTickets":
        result = ambilDataRekapTemuan(payload.statusFilter || "ALL");
        break;
        
      case "closeTicket":
        result = closeJobTicket(payload.ticketId, payload.picName, payload.photoBase64, payload.notes, { isDev: false });
        break;
        
      case "getGalleryPhotos":
        result = getFotoProsesInspeksi();
        break;
        
      case "getEmployees":
        result = getKaryawan();
        break;
        
      default:
        throw new Error("Action tidak valid atau belum diatur di GAS");
    }

    return ContentService.createTextOutput(JSON.stringify({ 
      success: true, 
      data: result 
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      error: error.message 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

    function compressImage(file, maxWidth = 600, maxHeight = 600, quality = 0.6) { 
        return new Promise((resolve, reject) => { 
            const reader = new FileReader(); 
            reader.readAsDataURL(file); 
            reader.onload = event => { 
                const img = new Image(); 
                img.src = event.target.result; 
                img.onload = () => { 
                    let width = img.width; 
                    let height = img.height; 
                    if (width > height) { 
                        if (width > maxWidth) { 
                            height = Math.round((height *= maxWidth / width)); 
                            width = maxWidth; 
                        } 
                    } else { 
                        if (height > maxHeight) { 
                            width = Math.round((width *= maxHeight / height)); 
                            height = maxHeight; 
                        } 
                    } 
                    const canvas = document.createElement('canvas'); 
                    canvas.width = width; 
                    canvas.height = height; 
                    const ctx = canvas.getContext('2d'); 
                    ctx.drawImage(img, 0, 0, width, height); 
                    // Pastikan output format diset ke image/jpeg
                    resolve(canvas.toDataURL('image/jpeg', quality)); 
                }; 
                img.onerror = error => reject(error); 
            }; 
            reader.onerror = error => reject(error); 
        }); 
    }

    function muatDraft() {
  const draftJSON = localStorage.getItem("draftInspeksiAll");
  if (!draftJSON) return;

  try {
    isMemuatDraft = true;
    const draft = JSON.parse(draftJSON);
    if (!draft.formVal) return;

    document.getElementById("pilihJenisInspeksi").value = draft.formVal;
    switchModeInspeksi();

    if (draft.tipe === "SARANA" && draft.jumlahSarana > 1) {
      for (let i = 1; i < draft.jumlahSarana; i++) tambahUnitSarana();
    }

    setTimeout(() => {
      // 1. Muat Data Tabel (APD) atau Form Dinamis
      if (draft.tipe === "APD") {
        if (draft.wkt) {
          document.getElementById("filterWaktu").value = draft.wkt;
          generateTable();
        }
        for (const [nama, data] of Object.entries(draft.tabelData)) {
          let idStr = nama.replace(/\s+/g, '');
          let row = document.getElementById("tr_" + idStr);
          if (!row) tambahkanBarisTabel(nama, "Karyawan Tambahan");
          row = document.getElementById("tr_" + idStr);
          if (row) {
            row.querySelector(".val-kehadiran").value = data.kehadiran;
            const cbs = row.querySelectorAll(".apd-cek");
            data.apd.forEach((isC, idx) => { cbs[idx].checked = isC; });
            row.querySelector(".val-ket").value = data.ket;
            updateRowStyle(nama);
          }
        }
      } else {
        if (draft.dynamicState && draft.dynamicState.length > 0) {
          document.querySelectorAll('#containerDynamic input:not([type="file"]), #containerDynamic select, #containerDynamic textarea').forEach((el, index) => {
            let state = draft.dynamicState[index];
            if (state) {
              if (state.type === 'checked') {
                el.checked = state.val;
                if (state.val && el.onchange) try { el.onchange(); } catch (e) {}
